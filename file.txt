CREATE TABLE public.barbers (
  id bigint NOT NULL DEFAULT nextval('barbers_id_seq'::regclass),
  shop_id uuid NOT NULL,
  name character varying NOT NULL,
  email character varying,
  phone character varying,
  specialties ARRAY,
  experience character varying,
  rating numeric DEFAULT 4.5,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  updated_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  CONSTRAINT barbers_pkey PRIMARY KEY (id),
  CONSTRAINT barbers_shop_id_fkey FOREIGN KEY (shop_id) REFERENCES public.shops(id)
);
CREATE TABLE public.bookings (
  id bigint NOT NULL DEFAULT nextval('bookings_id_seq'::regclass),
  customer_id uuid NOT NULL,
  shop_id uuid NOT NULL,
  barber_id bigint,
  service_ids ARRAY,
  booking_date date NOT NULL,
  booking_time time without time zone NOT NULL,
  status character varying DEFAULT 'pending'::character varying,
  total_price numeric,
  total_duration integer,
  customer_notes text,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  updated_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  completion_otp text,
  CONSTRAINT bookings_pkey PRIMARY KEY (id),
  CONSTRAINT bookings_customer_id_fkey FOREIGN KEY (customer_id) REFERENCES public.customers(id),
  CONSTRAINT bookings_shop_id_fkey FOREIGN KEY (shop_id) REFERENCES public.shops(id),
  CONSTRAINT bookings_barber_id_fkey FOREIGN KEY (barber_id) REFERENCES public.barbers(id)
);
CREATE TABLE public.customers (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  name character varying NOT NULL,
  email character varying NOT NULL UNIQUE,
  age integer,
  sex character varying,
  phone character varying,
  location character varying,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  updated_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  latitude double precision,
  longitude double precision,
  CONSTRAINT customers_pkey PRIMARY KEY (id),
  CONSTRAINT customers_id_fkey FOREIGN KEY (id) REFERENCES auth.users(id)
);
CREATE TABLE public.pending_bookings (
  order_id text NOT NULL,
  shop_id uuid NOT NULL,
  customer_id uuid NOT NULL,
  barber_id bigint NOT NULL,
  service_ids ARRAY NOT NULL,
  booking_date date NOT NULL,
  booking_time text NOT NULL,
  total_price numeric NOT NULL,
  total_duration integer NOT NULL,
  customer_notes text,
  status text NOT NULL DEFAULT 'PENDING'::text,
  created_at timestamp with time zone DEFAULT now(),
  final_amount real,
  CONSTRAINT pending_bookings_pkey PRIMARY KEY (order_id)
);
CREATE TABLE public.reviews (
  id bigint NOT NULL DEFAULT nextval('reviews_id_seq'::regclass),
  booking_id bigint NOT NULL,
  customer_id uuid NOT NULL,
  shop_id uuid NOT NULL,
  rating numeric NOT NULL CHECK (rating >= 0::numeric AND rating <= 5::numeric),
  feedback text,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  CONSTRAINT reviews_pkey PRIMARY KEY (id),
  CONSTRAINT fk_shop FOREIGN KEY (shop_id) REFERENCES public.shops(id)
);
CREATE TABLE public.services (
  id bigint NOT NULL DEFAULT nextval('services_id_seq'::regclass),
  shop_id uuid NOT NULL,
  name character varying NOT NULL,
  description text,
  duration integer NOT NULL,
  price numeric NOT NULL,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  updated_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  CONSTRAINT services_pkey PRIMARY KEY (id),
  CONSTRAINT services_shop_id_fkey FOREIGN KEY (shop_id) REFERENCES public.shops(id)
);
CREATE TABLE public.shop_images (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  shop_id uuid,
  image_url text NOT NULL,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  CONSTRAINT shop_images_pkey PRIMARY KEY (id),
  CONSTRAINT shop_images_shop_id_fkey FOREIGN KEY (shop_id) REFERENCES public.shops(id)
);
CREATE TABLE public.shop_owners (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  name character varying NOT NULL,
  email character varying NOT NULL UNIQUE,
  phone character varying,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  updated_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  vendor_id text,
  shop_id uuid UNIQUE,
  CONSTRAINT shop_owners_pkey PRIMARY KEY (id),
  CONSTRAINT shop_owners_id_fkey FOREIGN KEY (id) REFERENCES auth.users(id),
  CONSTRAINT shop_owners_shop_id_fkey FOREIGN KEY (shop_id) REFERENCES public.shops(id)
);
CREATE TABLE public.shop_settings (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  key text NOT NULL UNIQUE,
  value text,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT shop_settings_pkey PRIMARY KEY (id)
);
CREATE TABLE public.shops (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  name character varying NOT NULL,
  type character varying NOT NULL,
  email character varying,
  phone character varying,
  address text,
  is_open boolean DEFAULT true,
  opening_time time without time zone DEFAULT '09:00:00'::time without time zone,
  closing_time time without time zone DEFAULT '18:00:00'::time without time zone,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  updated_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  shop_latitude double precision,
  shop_longitude double precision,
  cashfree_vendor_id text,
  closed_from date,
  closed_to date,
  CONSTRAINT shops_pkey PRIMARY KEY (id)
);
